function emaArr(x, n) { const a = 2 / (n + 1), o = new Array(x.length); o[0] = x[0]; for (let i = 1; i < x.length; i++) o[i] = a * x[i] + (1 - a) * o[i - 1]; return o; }
function indicateurs15(B) {
  const c = B.map(b => b[3]), h = B.map(b => b[1]), l = B.map(b => b[2]), n = c.length, I = {};
  EMAS.forEach(e => I["ema" + e] = emaArr(c, e));
  I.bb_h = new Array(n).fill(null); I.bb_m = new Array(n).fill(null); I.bb_b = new Array(n).fill(null);
  for (let i = 19; i < n; i++) { let s = 0; for (let k = i - 19; k <= i; k++) s += c[k]; const m = s / 20; let v = 0; for (let k = i - 19; k <= i; k++) v += (c[k] - m) ** 2; const sd = Math.sqrt(v / 20);
    I.bb_m[i] = m; I.bb_h[i] = m + 2 * sd; I.bb_b[i] = m - 2 * sd; }
  I.rsi = new Array(n).fill(null); let au = 0, ad = 0;
  for (let i = 1; i < n; i++) { const d = c[i] - c[i - 1], up = Math.max(d, 0), dn = Math.max(-d, 0);
    if (i <= 14) { au += up / 14; ad += dn / 14; if (i === 14) I.rsi[i] = ad === 0 ? 100 : 100 - 100 / (1 + au / ad); }
    else { au = (au * 13 + up) / 14; ad = (ad * 13 + dn) / 14; I.rsi[i] = ad === 0 ? 100 : 100 - 100 / (1 + au / ad); } }
  const e12 = emaArr(c, 12), e26 = emaArr(c, 26); I.macd = c.map((_, i) => e12[i] - e26[i]); I.macd_s = emaArr(I.macd, 9); I.macd_h = I.macd.map((v, i) => v - I.macd_s[i]);
  I.atr = new Array(n).fill(null); let atr = 0;
  for (let i = 0; i < n; i++) { const pc = i ? c[i - 1] : c[0], tr = Math.max(h[i] - l[i], Math.abs(h[i] - pc), Math.abs(l[i] - pc));
    if (i < 13) atr += tr / 14; else if (i === 13) { atr += tr / 14; I.atr[i] = atr; } else { atr = (atr * 13 + tr) / 14; I.atr[i] = atr; } }
  const bw = c.map((_, i) => I.bb_m[i] == null ? null : (I.bb_h[i] - I.bb_b[i]) / I.bb_m[i]);
  I.bbw_rang = new Array(n).fill(null);
  for (let i = 0; i < n; i++) { if (bw[i] == null) continue; const w = []; for (let k = Math.max(0, i - 499); k <= i; k++) if (bw[k] != null) w.push(bw[k]);
    if (w.length < 100) continue; let moins = 0, egal = 0; w.forEach(v => { if (v < bw[i]) moins++; else if (v === bw[i]) egal++; }); I.bbw_rang[i] = (moins + (egal + 1) / 2) / w.length * 100; }
  return I;
}
function probaML(B, I) {
  const c = B.map(b => b[3]), n = c.length, lr = c.map((v, i) => i ? Math.log(v / c[i - 1]) : 0), p = new Array(n).fill(null);
  const somme = (i, k) => { if (i < k - 1) return null; let s = 0; for (let j = i - k + 1; j <= i; j++) s += lr[j]; return s; };
  for (let i = 0; i < n; i++) {
    const b = B[i][5], f = [lr[i], somme(i, 4), somme(i, 16), I.rsi[i] == null ? null : (I.rsi[i] - 50) / 50, I.macd_h[i] / c[i] * 100,
      I.bb_m[i] == null ? null : (c[i] - I.bb_m[i]) / (I.bb_h[i] - I.bb_m[i]), I.atr[i] == null ? null : I.atr[i] / c[i] * 100,
      I.bbw_rang[i] == null ? null : I.bbw_rang[i] / 100, (I.ema8[i] - I.ema55[i]) / c[i] * 100, Math.sin(2 * Math.PI * b / 26), Math.cos(2 * Math.PI * b / 26)];
    if (f.some(v => v == null || !isFinite(v))) continue;
    let z = MODELE.w[0]; f.forEach((v, k) => z += MODELE.w[k + 1] * (v - MODELE.mu[k]) / MODELE.sd[k]); p[i] = 1 / (1 + Math.exp(-z));
  }
  return p;
}

function tableaux(lignes, fin, F) {
  const O = new Array(390).fill(NaN), H = O.slice(), L = O.slice(), C = O.slice(), V = new Array(390).fill(0);
  lignes.forEach(([m, o, h, l, c, v]) => { if (m >= 0 && m <= fin) { O[m] = o; H[m] = h; L[m] = l; C[m] = c; V[m] = v || 0; } });
  const nb = C.filter(x => !isNaN(x)).length, bords = !isNaN(C[0]) && !isNaN(C[389]);
  for (let m = 1; m <= fin; m++) if (isNaN(C[m])) C[m] = C[m - 1];
  for (let m = fin - 1; m >= 0; m--) if (isNaN(C[m])) C[m] = C[m + 1];
  for (let m = 0; m <= fin; m++) { if (isNaN(O[m])) O[m] = C[m]; if (isNaN(H[m])) H[m] = Math.max(O[m], C[m]); if (isNaN(L[m])) L[m] = Math.min(O[m], C[m]); }
  return { O, H, L, C, V, fin, F, complete: F >= 390 && bords && nb >= 370 };
}

  if (!barre || m < 0 || m > 389) return;
  if (!src[jour]) { src[jour] = []; js.push(jour); js.sort(); }
  const L = src[jour], i = L.findIndex(r => r[0] === m), row = [m, ...barre];

}

// une ligne du journal d'un agent : "AAAA-MM-JJ HH:MM:SS UTC texte"

  const m = j.ligne.match(/^(\d{4})-(\d{2})-(\d{2}) (\d{2}):(\d{2}):(\d{2}) UTC (.*)$/); if (!m) return;
  const texte = m[7], nom = j.agent.replace(/^Bot3en1$/i, "Bot 3 en 1").replace(/^SupertrendNQ$/i, "Supertrend NQ").replace(/^XGBoostNQ$/i, "XGBoost NQ");
  const t = Date.UTC(+m[1], +m[2] - 1, +m[3], +m[4], +m[5], +m[6]), hp = parisHM(t);
  const pos = texte.match(/\(position (-?\d+) -> (-?\d+)\)/);
  const action = pos ? Math.sign(+pos[2]) : /^(achat|vente)\b/.test(texte) ? (texte.startsWith("achat") ? 1 : -1) : /^(sortie|fin de seance)|position MNQ fermee|arretee/.test(texte) ? 0 : null;

function executerSt(cible, p, motif) {
  const d = cible - ST.pos; if (!d) return;
  if (ST.pos !== 0 && Math.sign(cible) !== Math.sign(ST.pos)) { ST.real += ST.pos * (p - ST.prix) * 2 - Math.abs(ST.pos) * 3; ST.prix = cible ? p : 0; }
  else if (ST.pos !== 0 && Math.abs(cible) < Math.abs(ST.pos)) { const f = ST.pos - cible; ST.real += f * (p - ST.prix) * 2 - Math.abs(f) * 3; }
  else ST.prix = ST.pos ? (ST.prix * Math.abs(ST.pos) + p * Math.abs(d)) / Math.abs(cible) : p;
  ST.pos = cible; if (!cible) ST.prix = 0;
  ST.ordres.unshift({ t: parisHM(LIVE.utc), c: d > 0 ? "achat" : "vente", q: Math.abs(d), p, motif });
}
